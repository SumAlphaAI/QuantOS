import {bffZodSchemas,createBffClient} from '@sumalpha/api-client';
import {completeSecurityMfa,completeRecentAuth} from '../auth/bff';
import type {RouteContext} from '../auth/route-access';
import {approval,identity,refusal,reasonFor,type Approval,type Decision} from './policy';
export class ApprovalError extends Error{constructor(readonly status:number,readonly code='UNAVAILABLE',readonly correlationId?:string){super('Approval operation could not be verified');}}
const authority=(c:RouteContext)=>JSON.stringify([c.actorId,c.tenantId,c.workspaceId,c.accountId,c.mode,c.environment,c.mfaState,[...new Set(c.capabilities)].sort()]);
type Intent={approval:Approval;decision:Decision;key:string;body:{decision:Decision;reason?:string;mfaChallengeRef:string;reauthTokenRef:string};proofExpires:number};
export function createApprovalRuntime(config:{origin:string;authorize:()=>void;online:()=>boolean;invalidate:(status:number)=>void;csrf:()=>string;fetchImpl?:typeof fetch;now?:()=>number}){
 const now=config.now??Date.now,life=new AbortController(),options={signal:life.signal,timeoutMs:5000},client=createBffClient({baseUrl:config.origin,fetch:config.fetchImpl,...options}),listeners=new Set<()=>void>(),completed=new Set<string>();
 let context:RouteContext|undefined,rows:Approval[]=[],selected:Approval|undefined,receipt:Approval|undefined,pending:Intent|undefined,busy=false,retryAt=0,lastReadAt:string|undefined,nextCursor:string|undefined,seen=new Set<string>(),verifiedUntil=0,status:'empty'|'ready'|'changed'|'uncertain'|'error'='empty';
 const emit=()=>listeners.forEach(f=>f()),fail=(s:number,code:string):never=>{throw new ApprovalError(s,code);};
 function guard(){config.authorize();if(life.signal.aborted||!config.online()||!context||!['paper','shadow'].includes(context.mode)||!context.capabilities.includes('execution.operate'))fail(403,'FORBIDDEN');if(!Number.isFinite(now())||now()>=Date.parse(context!.expiresAt))fail(401,'EXPIRED');}
 function checked<T>(r:{response:Response;data?:T;error?:unknown}):T{config.authorize();if(life.signal.aborted)fail(403,'FORBIDDEN');if(r.response.status===200&&r.data!==undefined)return r.data;const e=bffZodSchemas.ErrorEnvelope.safeParse(r.error);if([401,403].includes(r.response.status))config.invalidate(r.response.status);if(r.response.status===429)retryAt=now()+Math.min(Math.max(e.success?e.data.retryAfter??60:60,1),3600)*1000;throw new ApprovalError(r.response.status,e.success?e.data.code:'UNAVAILABLE',e.success?e.data.correlationId:undefined);}
 async function load(){config.authorize();if(!config.origin||life.signal.aborted||!config.online())fail(503,'UNAVAILABLE');const a=bffZodSchemas.SessionContext.parse(checked(await client.GET('/v1/session',{cache:'no-store'}))),b=bffZodSchemas.SessionContext.parse(checked(await client.GET('/v1/context',{cache:'no-store'})));if(authority(a)!==authority(b)||context&&authority(context)!==authority(b)||b.mfaState==='challenged'){config.invalidate(403);fail(403,'FORBIDDEN');}context={...b,expiresAt:new Date(Math.min(Date.parse(a.expiresAt),Date.parse(b.expiresAt),context?Date.parse(context.expiresAt):Infinity)).toISOString()};guard();return structuredClone(context);}
 async function exclusive(fn:()=>Promise<void>){guard();if(busy)fail(409,'BUSY');if(now()<retryAt)fail(429,'RATE_LIMITED');busy=true;emit();try{await fn();}catch(e){const x=e as {status?:number;retryAfter?:number};if(x.status===429)retryAt=Math.max(retryAt,now()+Math.min(Math.max(x.retryAfter??60,1),3600)*1000);verifiedUntil=0;if(!pending&&status!=='changed')status='error';throw e;}finally{busy=false;emit();}}
 async function read(id:string){const a=approval(checked(await client.GET('/v1/approvals/{approvalId}',{params:{path:{approvalId:id}},cache:'no-store'})),id);guard();return a;}
 async function refresh(id:string){await load();selected=await read(id);verifiedUntil=now()+8000;lastReadAt=new Date(now()).toISOString();status=pending?'uncertain':'ready';}
 function writable(a=selected){const code=refusal(a,context,now());if(code)fail(409,code);if(completed.has(identity(a!)))fail(409,'ALREADY_RECORDED');}
 async function send(i:Intent,recovering:boolean){guard();if(now()>=i.proofExpires||now()>=Date.parse(i.approval.expiresAt))fail(409,'RECOVERY_EXPIRED');let dispatched=false;
 try{dispatched=true;const r=await client.POST('/v1/approvals/{approvalId}/decisions',{params:{path:{approvalId:i.approval.approvalId},header:{'If-Match':i.approval.objectVersion,'Idempotency-Key':i.key,'X-Request-Id':crypto.randomUUID(),'X-CSRF-Token':config.csrf()}},body:i.body});
 if(r.response.status!==200){const known=bffZodSchemas.ErrorEnvelope.safeParse(r.error).success&&[401,403,409,422,429].includes(r.response.status);if(known&&!recovering){pending=undefined;selected=undefined;}checked(r);}
 const a=approval(checked(r),i.approval.approvalId);guard();if(a.originator!==i.approval.originator||a.correlationId!==i.approval.correlationId||a.objectVersion===i.approval.objectVersion||a.decision!==i.decision||a.status!==(i.decision==='approve'?'approved':'rejected'))fail(502,'RESPONSE_BINDING');
 completed.add(identity(i.approval));receipt=a;selected=a;pending=undefined;verifiedUntil=0;status='ready';
 }catch(e){if(pending&&dispatched)status='uncertain';throw e;}}
 return{load,subscribe(fn:()=>void){listeners.add(fn);return()=>{listeners.delete(fn);};},get view(){const blocked=refusal(selected,context,now());return structuredClone({context,rows,selected,receipt,busy,retryAt,lastReadAt,nextCursor,status,pending:!!pending,pendingDecision:pending?.decision,canRecover:!!pending&&now()<pending.proofExpires&&now()<Date.parse(pending.approval.expiresAt),refusal:blocked,writable:!blocked&&!pending&&!busy&&now()<verifiedUntil&&!completed.has(selected?identity(selected):'')});},
 list(more=false){return exclusive(async()=>{await load();if(more&&(!nextCursor||seen.has(nextCursor)||rows.length>=1000))fail(409,'PAGE_LIMIT');const cursor=more?nextCursor:undefined,page=bffZodSchemas.Page.parse(checked(await client.GET('/v1/approvals',{params:{query:{pageSize:50,...(cursor?{cursor}:{})}},cache:'no-store'})));guard();if(page.items.length>50)fail(502,'PAGE_LIMIT');const next=page.items.map(v=>approval(v)),ids=new Set(more?rows.map(a=>a.approvalId):[]);for(const a of next){if(ids.has(a.approvalId))fail(502,'DUPLICATE');ids.add(a.approvalId);}if(page.nextCursor&&(page.nextCursor===cursor||more&&seen.has(page.nextCursor)||page.nextCursor.length>2048))fail(502,'CURSOR_LOOP');if(!more)seen=new Set();if(cursor)seen.add(cursor);rows=more?[...rows,...next]:next;nextCursor=page.nextCursor;lastReadAt=new Date(now()).toISOString();status=pending?'uncertain':'ready';});},
 select(id:string){bffZodSchemas.UUID.parse(id);return exclusive(async()=>{if(pending&&selected?.approvalId!==id)fail(409,'RECOVERY_REQUIRED');selected=undefined;verifiedUntil=0;await refresh(id);});},
 act(decision:Decision,reason:string,phrase:string,code:string){return exclusive(async()=>{writable();if(pending||completed.size>=100)fail(409,'RECOVERY_REQUIRED');if(!['approve','reject','request_info'].includes(decision)||phrase!=='CONFIRM '+decision.toUpperCase()||!/^\d{6}$/.test(code))fail(422,'INVALID_INPUT');const why=reasonFor(decision,reason),reviewed=selected!;await refresh(reviewed.approvalId);writable();if(identity(selected!)!==identity(reviewed)){status='changed';fail(409,'SOURCE_CHANGED');}
 const m=await completeSecurityMfa(config.origin,code,config.csrf(),config.fetchImpl,options);guard();if(m.status!=='verified')fail(403,'MFA_REQUIRED');const g=await completeRecentAuth(config.origin,m.challengeRef,config.csrf(),config.fetchImpl,options);guard();const latest=await read(reviewed.approvalId);await load();writable(latest);if(identity(latest)!==identity(reviewed)){selected=latest;status='changed';fail(409,'SOURCE_CHANGED');}if(Date.parse(g.expiresAt)<=now())fail(403,'MFA_EXPIRED');pending={approval:reviewed,decision,key:crypto.randomUUID(),proofExpires:Date.parse(g.expiresAt),body:{decision,...(why?{reason:why}:{}),mfaChallengeRef:m.challengeRef,reauthTokenRef:g.reauthTokenRef}};await send(pending,false);});},
 retry(){return exclusive(async()=>{if(!pending)fail(409,'RECOVERY_REQUIRED');await load();await send(pending!,true);});},
 close(){life.abort();context=undefined;selected=undefined;receipt=undefined;rows=[];pending=undefined;completed.clear();nextCursor=undefined;seen.clear();lastReadAt=undefined;verifiedUntil=0;listeners.clear();}
 };
}
export type ApprovalRuntime=ReturnType<typeof createApprovalRuntime>;
