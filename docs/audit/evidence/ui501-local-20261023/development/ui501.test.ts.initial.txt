import { describe, it, expect } from 'vitest';
import { visibleShellNavigation } from '@sumalpha/domain-ui/ui101';
import { commandNavigationCapabilities } from '../src/command/presentation';
import { money, decimal, snapshot, fresh, PortfolioFacts, riskAlerts } from '../src/portfolio/policy';
import { createPortfolioRuntime } from '../src/portfolio/runtime';
import type { RouteContext } from '../src/auth/route-access';
const id='11111111-1111-4111-8111-111111111111', other='22222222-2222-4222-8222-222222222222', now=Date.parse('2026-10-23T00:00:00Z');
const ctx:RouteContext={actorId:id,tenantId:id,workspaceId:id,accountId:id,mode:'paper',environment:'dev',capabilities:['execution.operate','ops.read'],mfaState:'verified',expiresAt:new Date(now+600000).toISOString()};
const pair=()=>({portfolio:{accountId:id,asOf:new Date(now).toISOString(),stale:false,positions:[{symbol:'BTC/USDT',side:'long' as const,quantity:'9007199254740993.123456789',avgPrice:'100.000000001'}],pnl:{currencyCode:'USD',units:'-9007199254740993',nanos:-123456789},exposure:'9007199254740993.123456789'},risk:{accountId:id,asOf:new Date(now).toISOString(),hitRules:[] as string[],limitIds:[] as string[],killSwitch:{state:'disengaged' as 'engaged'|'disengaged'},objectVersion:'0'}});
const event=(value=pair(),sequence=1)=>({streamId:id,sequence,eventId:id,occurredAt:new Date(now).toISOString(),correlationId:id,payloadVersion:'v1' as const,payload:{type:'portfolio',...value}});
const error=(status:number,code='INVALID_REQUEST',retryAfter?:number)=>Response.json({code,message:'private SQL must never display',correlationId:id,...retryAfter?{retryAfter}:{}},{status});
function setup(handler?:(r:Request)=>Promise<Response|void>|Response|void) {
 let clock=now,online=true,authorized=true,csrf='csrf',context={...ctx},value=pair(),journal=[event(value)],readFailure=false;
 const requests:Request[]=[],invalidations:number[]=[];
 const runtime=createPortfolioRuntime({origin:'https://bff.example',now:()=>clock,online:()=>online,csrf:()=>csrf,authorize:()=>{if(!authorized)throw Error('revoked');},invalidate:s=>invalidations.push(s),fetchImpl:async(input,init)=>{
 const r=new Request(input,init);requests.push(r.clone());const custom=await handler?.(r);if(custom)return custom;const path=new URL(r.url).pathname;
 if(['/v1/session','/v1/context'].includes(path))return readFailure?error(503,'UNAVAILABLE'):Response.json(context);
 if(path==='/v1/portfolio')return Response.json(value.portfolio);if(path==='/v1/risk')return Response.json(value.risk);
 if(path==='/v1/alerts')return Response.json({items:[],nextCursor:null});
 if(path.endsWith('/stream')){const after=Number(new URL(r.url).searchParams.get('afterSequence'));return new Response(journal.filter(e=>e.sequence>after).map(e=>`data: ${JSON.stringify(e)}\n\n`).join(''),{headers:{'content-type':'text/event-stream'}});}
 if(path.endsWith('/challenges'))return Response.json({challengeRef:id,status:'verified'});
 if(path.endsWith('/reauth'))return Response.json({reauthTokenRef:other,expiresAt:new Date(clock+30000).toISOString()});
 if(path.includes('/kill-switch/')){const body=await r.json();value={...value,risk:{...value.risk,killSwitch:{state:path.endsWith('/engage')?'engaged':'disengaged'},objectVersion:String(BigInt(value.risk.objectVersion)+1n),hitRules:path.endsWith('/engage')?['AccountKill']:[],limitIds:path.endsWith('/engage')?[id]:[]}};journal.push(event(value,journal.length+1));return Response.json({...value.risk,auditRef:id,correlationId:id,...body.reason?{}:{}});}
 throw Error(path);
 }});
 return {runtime,requests,invalidations,init:async()=>{await runtime.load();await runtime.refresh();},advance:(ms:number)=>{clock+=ms;},block:(v:string)=>{if(v==='offline')online=false;if(v==='revoked')authorized=false;if(v==='csrf')csrf='';if(v==='expired')clock+=600000;},setContext:(c:Partial<RouteContext>)=>{context={...context,...c};},setValue:(v:ReturnType<typeof pair>)=>{value=v;journal=[event(v)];},readFailure:(v:boolean)=>{readFailure=v;}};
}
const writes=(s:ReturnType<typeof setup>)=>s.requests.filter(r=>r.url.includes('/kill-switch/')&&r.method==='POST');
const act=(s:ReturnType<typeof setup>,action:'engage'|'release'='engage')=>s.runtime.act(action,'owned safety test',action==='engage'?'ENGAGE KILL SWITCH':'RELEASE KILL SWITCH','123456');
describe('UI501 exact C06 facts',()=>{
 it('formats signed money and all nine fractional digits without float loss',()=>{expect(money(pair().portfolio.pnl)).toBe('-9007199254740993.123456789 USD');expect(money({currencyCode:'USD',units:'0',nanos:-1})).toBe('-0.000000001 USD');expect(money({currencyCode:'USD',units:'12',nanos:100000000})).toBe('12.1 USD');expect(decimal('9007199254740993.123456789')).toContain('123456789');});
 it.each(['1e5','NaN','Infinity','01.2','1.','-1','0x10'])('rejects unsafe/noncanonical nonnegative decimal %s',v=>expect(()=>decimal(v,true)).toThrow());
 it.each([{units:'1',nanos:-1},{units:'-1',nanos:1},{units:'0',nanos:1e9},{units:'0',nanos:.1},{units:'NaN',nanos:0}])('rejects invalid MoneyValue %j',v=>expect(()=>money({...v,currencyCode:'USD'})).toThrow());
 it('validates account, coherent asOf, unique positions/rules, known decimals and kill agreement',()=>{for(const mutate of [(p:ReturnType<typeof pair>)=>{p.portfolio.accountId=other;},(p:ReturnType<typeof pair>)=>{p.risk.asOf=new Date(now+1).toISOString();},(p:ReturnType<typeof pair>)=>{p.portfolio.positions.push(p.portfolio.positions[0]);},(p:ReturnType<typeof pair>)=>{p.risk.killSwitch.state='engaged';},(p:ReturnType<typeof pair>)=>{p.risk.hitRules=['AccountKill'];},(p:ReturnType<typeof pair>)=>{p.risk.objectVersion='NaN';}]){const p=pair();mutate(p);expect(()=>snapshot(p,id,now)).toThrow();}expect(()=>snapshot(pair(),id,now-1)).toThrow();});
 it('expires strictly at sixty seconds and honors explicit stale',()=>{const p=pair();expect(fresh(p,now+59999)).toBe(true);expect(fresh(p,now+60000)).toBe(false);p.portfolio.stale=true;expect(fresh(p,now)).toBe(false);expect(fresh(undefined,now)).toBe(false);});
 it('atomically applies dense broadcasts, clones, ignores duplicates and rejects gaps, leases, rollback and late revoke',()=>{const f=new PortfolioFacts(id),e=event();expect(f.apply(e,now)).toBe(true);expect(f.apply(e,now)).toBe(false);const copy=f.current!;copy.risk.killSwitch.state='engaged';expect(f.current!.risk.killSwitch.state).toBe('disengaged');expect(()=>f.apply({...e,sequence:3},now)).toThrow('gap');expect(()=>f.apply({...e,sequence:2,streamId:other},now)).toThrow();const v=pair();v.risk.objectVersion='1';v.risk.killSwitch.state='engaged';v.risk.hitRules=['AccountKill'];v.risk.limitIds=[id];f.apply(event(v,2),now);expect(f.current!.risk.killSwitch.state).toBe('engaged');expect(()=>f.apply(event(pair(),3),now)).toThrow('regressed');expect(()=>f.apply({...e,sequence:0,payload:{type:'permission_revoked'}},now)).toThrow();expect(f.current).toBeUndefined();expect(()=>f.seed(pair(),now)).toThrow();});
 it('refuses foreign/mixed C16 data and never maps C06 hits into alerts',()=>{expect(riskAlerts([],ctx)).toEqual([]);expect(()=>riskAlerts([{domain:'system'}],ctx)).toThrow();expect(visibleShellNavigation(commandNavigationCapabilities(['execution.operate'])).map(v=>v.route)).toContain('/portfolio');expect(visibleShellNavigation(commandNavigationCapabilities(['settings.read'])).map(v=>v.route)).not.toContain('/portfolio');});
});
describe('UI501 bounded runtime',()=>{
 it('loads scoped GET/stream/alerts, exact amounts, persists only checked receipt and re-reads kill fact',async()=>{const s=setup();await s.init();expect(s.runtime.view.writable).toBe(true);expect(s.runtime.view.snapshot!.portfolio.positions[0].quantity).toBe(pair().portfolio.positions[0].quantity);await act(s);expect(s.runtime.view.snapshot!.risk.killSwitch.state).toBe('engaged');expect(s.runtime.view.receipt?.auditRef).toBe(id);expect(s.runtime.view.pending).toBe(false);expect(await writes(s)[0].json()).toEqual({accountId:id,reason:'owned safety test',confirmPhrase:'ENGAGE KILL SWITCH',mfaChallengeRef:id,reauthTokenRef:other});expect(writes(s)[0].headers.get('if-match')).toBeNull();expect(s.requests.filter(r=>r.method==='GET').every(r=>r.cache==='no-store')).toBe(true);expect(new URL(s.requests.find(r=>r.url.includes('/alerts'))!.url).searchParams.get('domain')).toBe('risk');await act(s,'release');expect(s.runtime.view.snapshot!.risk.killSwitch.state).toBe('disengaged');expect(await writes(s)[1].json()).not.toHaveProperty('confirmPhrase');});
 it.each(['offline','revoked','csrf','expired'])('blocks %s before MFA or risk writes',async v=>{const s=setup();await s.init();s.block(v);await expect(act(s)).rejects.toThrow();expect(s.requests.some(r=>r.method==='POST')).toBe(false);});
 it('blocks stale, silent stream expiry, future/inconsistent snapshots and role loss before dispatch',async()=>{const s=setup();await s.init();s.advance(8000);await expect(act(s)).rejects.toThrow();expect(writes(s)).toHaveLength(0);await s.runtime.poll();expect(s.runtime.view.writable).toBe(true);s.advance(60000);await s.runtime.poll();expect(s.runtime.view.writable).toBe(false);await expect(act(s)).rejects.toThrow();const b=setup();await b.init();b.setContext({capabilities:['settings.read']});await expect(act(b)).rejects.toThrow();expect(b.invalidations).toEqual([403]);expect(writes(b)).toHaveLength(0);});
 it('explicit server stale flag blocks every write including MFA',async()=>{const s=setup();const v=pair();v.portfolio.stale=true;s.setValue(v);await s.init();expect(s.runtime.view.writable).toBe(false);await expect(act(s)).rejects.toThrow();expect(s.requests.some(r=>r.method==='POST')).toBe(false);});
 it('validates reason, confirmation, six-digit MFA and state before dispatch',async()=>{const s=setup();await s.init();for(const args of [['','ENGAGE KILL SWITCH','123456'],['reason','wrong','123456'],['reason','ENGAGE KILL SWITCH','123'],['x'.repeat(1001),'ENGAGE KILL SWITCH','123456']])await expect(s.runtime.act('engage',args[0],args[1],args[2])).rejects.toThrow();await expect(act(s,'release')).rejects.toThrow();expect(s.requests.some(r=>r.method==='POST')).toBe(false);});
 it('lost reply locks intent; no read reconnect replay; explicit recovery retains original proof, body and key',async()=>{let lost=true;const s=setup(r=>{if(r.url.includes('/kill-switch/')&&lost){lost=false;throw Error('owned lost reply');}});await s.init();await expect(act(s)).rejects.toThrow();expect(s.runtime.view.pending).toBe(true);await expect(act(s)).rejects.toThrow();s.readFailure(true);await expect(s.runtime.retry()).rejects.toThrow();expect(writes(s)).toHaveLength(1);s.readFailure(false);await s.runtime.refresh();expect(writes(s)).toHaveLength(1);await s.runtime.retry();expect(s.runtime.view.pending).toBe(false);expect(writes(s)[0].headers.get('idempotency-key')).toBe(writes(s)[1].headers.get('idempotency-key'));expect(await writes(s)[0].json()).toEqual(await writes(s)[1].json());expect(s.requests.filter(r=>r.url.includes('/challenges'))).toHaveLength(1);});
 it.each([403,409,422,429])('first definite %s refusal clears pending, never fabricates receipt or kill state',async status=>{const s=setup(r=>r.url.includes('/kill-switch/')?error(status,status===429?'RATE_LIMITED':'INVALID_REQUEST',2):undefined);await s.init();await expect(act(s)).rejects.toMatchObject({status});expect(s.runtime.view.pending).toBe(false);expect(s.runtime.view.receipt).toBeUndefined();expect(s.runtime.view.snapshot!.risk.killSwitch.state).toBe('disengaged');if(status===429){await expect(act(s)).rejects.toThrow();expect(writes(s)).toHaveLength(1);}});
 it('unknown recovery refusal retains intent, and cooldown blocks new or retried dispatch',async()=>{let n=0;const s=setup(r=>{if(!r.url.includes('/kill-switch/'))return;if(n++===0)throw Error('lost');return error(429,'RATE_LIMITED',2);});await s.init();await expect(act(s)).rejects.toThrow();await expect(s.runtime.retry()).rejects.toThrow();expect(s.runtime.view.pending).toBe(true);await expect(s.runtime.retry()).rejects.toThrow();expect(writes(s)).toHaveLength(2);s.advance(2000);await expect(s.runtime.retry()).rejects.toThrow();expect(writes(s)).toHaveLength(3);expect(writes(s).every(r=>r.headers.get('idempotency-key')===writes(s)[0].headers.get('idempotency-key'))).toBe(true);});
 it('bad success receipt is outcome unknown and cannot update current risk',async()=>{const s=setup(r=>r.url.includes('/kill-switch/')?Response.json({...pair().risk,accountId:other,killSwitch:{state:'engaged'}}):undefined);await s.init();await expect(act(s)).rejects.toThrow();expect(s.runtime.view.pending).toBe(true);expect(s.runtime.view.snapshot!.risk.killSwitch.state).toBe('disengaged');});
 it('failed MFA/expired reauth cannot reach C06',async()=>{for(const path of ['challenges','reauth']){const s=setup(r=>r.url.endsWith('/'+path)?Response.json(path==='challenges'?{challengeRef:id,status:'failed'}:{reauthTokenRef:other,expiresAt:new Date(now).toISOString()}):undefined);await s.init();await expect(act(s)).rejects.toThrow();expect(writes(s)).toHaveLength(0);}});
 it('stream failure freezes last snapshot, blocks all writes, manual refresh restores scoped cursor',async()=>{let broken=false;const s=setup(r=>broken&&r.url.includes('/portfolio/stream')?error(503,'UNAVAILABLE'):undefined);await s.init();broken=true;await expect(s.runtime.poll()).rejects.toThrow();expect(s.runtime.view.writable).toBe(false);expect(s.runtime.view.snapshot).toEqual(pair());await expect(act(s)).rejects.toThrow();broken=false;await s.runtime.refresh();expect(s.runtime.view.writable).toBe(true);s.runtime.close();expect(s.runtime.view.snapshot).toBeUndefined();expect(s.runtime.view.receipt).toBeUndefined();await expect(s.runtime.load()).rejects.toThrow();});
 it('unavailable/unauthorized C16 does not mean zero alerts and cannot add authority',async()=>{const s=setup(r=>r.url.includes('/alerts')?error(503,'UNAVAILABLE'):undefined);await s.init();expect(s.runtime.view.alertStatus).toBe('unavailable');expect(s.runtime.view.alerts).toEqual([]);const b=setup();b.setContext({capabilities:['execution.operate']});await b.init();expect(b.runtime.view.alertStatus).toBe('forbidden');expect(b.requests.some(r=>r.url.includes('/alerts'))).toBe(false);});
 it('concurrent actions use one MFA/intent and no late response can resurrect a closed runtime',async()=>{let unblock!:()=>void;const gate=new Promise<void>(r=>{unblock=r;});const s=setup(async r=>{if(r.url.endsWith('/challenges'))await gate;});await s.init();const pending=act(s);await expect(act(s)).rejects.toThrow();s.runtime.close();unblock();await expect(pending).rejects.toThrow();expect(writes(s)).toHaveLength(0);expect(s.runtime.view.snapshot).toBeUndefined();});
});
