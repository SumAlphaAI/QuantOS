function setup(handler?:(r:Request)=>Response|void|Promise<Response|void>){let clock=now,p=proposal(),caps=['execution.operate','audit:read'];const requests:Request[]=[],invalidations:number[]=[];const runtime=createProposalRuntime({origin:'https://bff.example',now:()=>clock,authorize:()=>{},online:()=>true,csrf:()=>'',invalidate:n=>invalidations.push(n),fetchImpl:async(input,init)=>{const r=new Request(input,init);requests.push(r.clone());const custom=await handler?.(r);if(custom)return custom;const path=new URL(r.url).pathname;if(path==='/v1/session'||path==='/v1/context')return Response.json({actorId:id,tenantId:id,workspaceId:id,accountId:id,mode:'paper',environment:'dev',capabilities:caps,mfaState:'verified',expiresAt:stamp(600000)});if(path.endsWith('/stream'))return new Response('data: '+JSON.stringify({streamId:id,sequence:1,eventId:other,occurredAt:stamp(),correlationId:p.correlationId,payloadVersion:'v1',payload:p})+'\n\n',{headers:{'content-type':'text/event-stream'}});if(path.startsWith('/v1/audit/'))return Response.json(page({correlationId:p.correlationId}));return Response.json(p);}});return{runtime,requests,invalidations,init:async()=>{await runtime.load();await runtime.open(id);},set:(v:Proposal)=>{p=v;},caps:(v:string[])=>{caps=v;},advance:(ms:number)=>{clock+=ms;}};}
const reads=(s:ReturnType<typeof setup>)=>s.requests.filter(r=>r.url.includes('/audit/'));
